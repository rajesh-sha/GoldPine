type LiqLevel
    string name
    string kind
    bool   enabled
    color  col
    float  price   = na
    bool   broke   = false
    float  extreme = na
    int    barsOut = 0
    bool   armed   = true
    int    born    = 0
    line   ln      = na
    label  lb      = na
    string pool    = ""   // "BSL" buy-side / "SSL" sell-side / "RND"

method refresh(LiqLevel L, float newPrice) =>
    L.price   := newPrice
    L.broke   := false
    L.extreme := na
    L.barsOut := 0
    L.armed   := true
    if not na(L.ln)
        line.delete(L.ln)
    if not na(L.lb)
        label.delete(L.lb)
    if L.enabled and not na(newPrice)
        string side = L.kind == "high" ? "BSL" : "SSL"
        L.pool := side
        string tip = not useSimple ? " [" + side + "]" : L.kind == "high" ? " | fake breakout UP = SELL (BSL swept)" : " | fake breakout DOWN = BUY (SSL swept)"
        L.ln := line.new(bar_index, newPrice, bar_index + 1, newPrice, extend = extend.right, color = L.col, width = 2)
        L.lb := label.new(bar_index, newPrice, L.name + "  " + str.tostring(newPrice, format.mintick) + tip, style = label.style_label_right, color = color.new(L.col, 80), textcolor = L.col, size = size.small)

method track(LiqLevel L, float minPen, int window) =>
    bool bearSweep = false
    bool bullSweep = false
    bool bullBreak = false
    bool bearBreak = false
    string sweptName = ""
    if L.enabled and L.armed and not na(L.price)
        if L.kind == "high"
            if high > L.price
                L.extreme := L.broke ? math.max(L.extreme, high) : high
                L.broke   := true
                if close > L.price
                    L.barsOut += 1
            if L.broke and close < L.price and (L.extreme - L.price) >= minPen and L.barsOut <= window
                bearSweep := true
                sweptName := L.name
                L.broke   := false
                L.armed   := reArm
            else if L.broke and L.barsOut > window and close > L.price
                bullBreak := true
                L.broke   := false
                L.armed   := false
        else
            if low < L.price
                L.extreme := L.broke ? math.min(L.extreme, low) : low
                L.broke   := true
                if close < L.price
                    L.barsOut += 1
            if L.broke and close > L.price and (L.price - L.extreme) >= minPen and L.barsOut <= window
                bullSweep := true
                sweptName := L.name
                L.broke   := false
                L.armed   := reArm
            else if L.broke and L.barsOut > window and close < L.price
                bearBreak := true
                L.broke   := false
                L.armed   := false
    [bearSweep, bullSweep, bullBreak, bearBreak, sweptName]

    L.ln := line.new(bar_index, px, bar_index + 1, px, extend = extend.right, color = color.new(col, 35), width = 1, style = line.style_dotted)
    L.lb := label.new(bar_index, px, nm + tip, style = label.style_label_right, color = color.new(col, 88), textcolor = col, size = size.tiny)
    array.push(roundLevels, L)

var float lastRoundAnchor = na
if useRound and barstate.isconfirmed
    float anchor = math.round(close / roundStep) * roundStep
    bool needRebuild = na(lastRoundAnchor) or math.abs(anchor - lastRoundAnchor) >= roundStep or newDay
    if needRebuild
        f_clearRounds()
        for i = 1 to roundCount
            f_addRound(anchor + i * roundStep, true)
            f_addRound(anchor - i * roundStep, false)
        lastRoundAnchor := anchor

// ═══════════════════════════ EQH / EQL DYNAMIC POOLS ═══════════════════════════
f_addDyn(nm, kind, px, col) =>
    if array.size(dynLevels) >= maxDyn
        LiqLevel old = array.shift(dynLevels)
        if not na(old.ln)
            line.delete(old.ln)
        if not na(old.lb)
            label.delete(old.lb)
    LiqLevel L = LiqLevel.new(nm, kind, true, col)
    L.price := px
    L.born  := bar_index
    L.pool  := kind == "high" ? "BSL" : "SSL"
    string dtip = not useSimple ? nm + " [" + L.pool + "]" : nm + (kind == "high" ? " double top | fake break UP = SELL" : " double bottom | fake break DOWN = BUY")
    L.ln    := line.new(bar_index - pivLen, px, bar_index, px, extend = extend.right, color = col, width = 1, style = line.style_dashed)
    L.lb    := label.new(bar_index, px, dtip, style = label.style_label_right, color = color.new(col, 85), textcolor = col, size = size.tiny)
    array.push(dynLevels, L)

ph = ta.pivothigh(high, pivLen, pivLen)
pl = ta.pivotlow(low, pivLen, pivLen)
var float lastPH = na
var float lastPL = na
if useEQ and not na(ph)
    if not na(lastPH) and math.abs(ph - lastPH) <= eqTolATR * atrVal
        f_addDyn("EQH", "high", math.max(ph, lastPH), color.maroon)
    lastPH := ph
if useEQ and not na(pl)
    if not na(lastPL) and math.abs(pl - lastPL) <= eqTolATR * atrVal
        f_addDyn("EQL", "low", math.min(pl, lastPL), color.navy)
    lastPL := pl

if array.size(dynLevels) > 0
    for i = array.size(dynLevels) - 1 to 0
        LiqLevel L = array.get(dynLevels, i)
        if (not L.armed) or (bar_index - L.born > dynAge)
            if not na(L.ln)
                line.delete(L.ln)
            if not na(L.lb)
                label.delete(L.lb)
            array.remove(dynLevels, i)

// ═══════════════════════════ SWING STRUCTURE (for MSS) ═══════════════════════════
mssPH = ta.pivothigh(high, mssPivot, mssPivot)
mssPL = ta.pivotlow(low, mssPivot, mssPivot)
var float swingHi = na
var float swingLo = na
var int   swingHiBar = na
var int   swingLoBar = na
if not na(mssPH)
    swingHi := mssPH
    swingHiBar := bar_index - mssPivot
if not na(mssPL)
    swingLo := mssPL
    swingLoBar := bar_index - mssPivot

// ═══════════════════════════ SWEEP SCAN ═══════════════════════════
f_scan(arr) =>
    bool  aBull = false
    bool  aBear = false
    float refL  = na
    float refS  = na
    bool  cL    = false
    bool  cS    = false
    float cLpx  = na
    float cSpx  = na
    string nBull = ""
    string nBear = ""
    if array.size(arr) > 0
        for L in arr
            [bearSw, bullSw, bullBrk, bearBrk, sn] = L.track(minSweepPts, confB)
            if bullSw
                aBull := true
                refL  := na(refL) ? L.extreme : math.min(refL, L.extreme)
                nBull := sn
            if bearSw
                aBear := true
                refS  := na(refS) ? L.extreme : math.max(refS, L.extreme)
                nBear := sn
            if bullBrk
                cL   := true
                cLpx := na(cLpx) ? L.price : math.max(cLpx, L.price)
            if bearBrk
                cS   := true
                cSpx := na(cSpx) ? L.price : math.min(cSpx, L.price)
    [aBull, aBear, refL, refS, cL, cS, cLpx, cSpx, nBull, nBear]

[sBull, sBear, sRefL, sRefS, sCL, sCS, sCLpx, sCSpx, sNBull, sNBear] = f_scan(staticLevels)
[dBull, dBear, dRefL, dRefS, dCL, dCS, dCLpx, dCSpx, dNBull, dNBear] = f_scan(dynLevels)
[rBull, rBear, rRefL, rRefS, rCL, rCS, rCLpx, rCSpx, rNBull, rNBear] = f_scan(roundLevels)

anyBull       = sBull or dBull or rBull
anyBear       = sBear or dBear or rBear
sweepRefLong  = na(sRefL) ? (na(dRefL) ? rRefL : na(rRefL) ? dRefL : math.min(dRefL, rRefL)) : (na(dRefL) ? (na(rRefL) ? sRefL : math.min(sRefL, rRefL)) : (na(rRefL) ? math.min(sRefL, dRefL) : math.min(sRefL, math.min(dRefL, rRefL))))
sweepRefShort = na(sRefS) ? (na(dRefS) ? rRefS : na(rRefS) ? dRefS : math.max(dRefS, rRefS)) : (na(dRefS) ? (na(rRefS) ? sRefS : math.max(sRefS, rRefS)) : (na(rRefS) ? math.max(sRefS, dRefS) : math.max(sRefS, math.max(dRefS, rRefS))))
contLong      = sCL or dCL or rCL
contShort     = sCS or dCS or rCS
contLongBase  = na(sCLpx) ? (na(dCLpx) ? rCLpx : na(rCLpx) ? dCLpx : math.max(dCLpx, rCLpx)) : (na(dCLpx) ? (na(rCLpx) ? sCLpx : math.max(sCLpx, rCLpx)) : (na(rCLpx) ? math.max(sCLpx, dCLpx) : math.max(sCLpx, math.max(dCLpx, rCLpx))))
contShortBase = na(sCSpx) ? (na(dCSpx) ? rCSpx : na(rCSpx) ? dCSpx : math.min(dCSpx, rCSpx)) : (na(dCSpx) ? (na(rCSpx) ? sCSpx : math.min(sCSpx, rCSpx)) : (na(rCSpx) ? math.min(sCSpx, dCSpx) : math.min(sCSpx, math.min(dCSpx, rCSpx))))
sweptLongName  = sBull ? sNBull : dBull ? dNBull : rNBull
sweptShortName = sBear ? sNBear : dBear ? dNBear : rNBear

// Sweep quality: rejection wick share of candle range
candleRange = high - low
upperWick = high - math.max(open, close)
lowerWick = math.min(open, close) - low
wickRatioBull = candleRange > 0 ? lowerWick / candleRange : 0.0
wickRatioBear = candleRange > 0 ? upperWick / candleRange : 0.0
qualityBull = wickRatioBull >= minWickRatio
qualityBear = wickRatioBear >= minWickRatio

// ═══════════════════════════ SESSION HIERARCHY ═══════════════════════════
inLondon  = useLondonKZ and not na(time(timeframe.period, londonKZ, TZ_NY))
inNY      = useNYKZ     and not na(time(timeframe.period, nyKZ, TZ_NY))
inSB      = useSB       and not na(time(timeframe.period, sbKZ, TZ_NY))
sessionOK = not useSessFilter or inLondon or inNY or inSB

var bool asiaHighSweptLondon = false
var bool asiaLowSweptLondon  = false
var bool londonHighSweptNY   = false
var bool londonLowSweptNY    = false
var float londonH = na
var float londonL = na

if inLondon and not inLondon[1]
    londonH := high
    londonL := low
    asiaHighSweptLondon := false
    asiaLowSweptLondon := false
else if inLondon
    londonH := math.max(nz(londonH, high), high)
    londonL := math.min(nz(londonL, low), low)
    if useAsia and not na(lvlAsiaH.price) and high > lvlAsiaH.price and close < lvlAsiaH.price
        asiaHighSweptLondon := true
    if useAsia and not na(lvlAsiaL.price) and low < lvlAsiaL.price and close > lvlAsiaL.price
        asiaLowSweptLondon := true

if inNY and not inNY[1]
    londonHighSweptNY := false
    londonLowSweptNY := false
if inNY
    if not na(londonH) and high > londonH and close < londonH
        londonHighSweptNY := true
    if not na(londonL) and low < londonL and close > londonL
        londonLowSweptNY := true

hierLong  = (inLondon and asiaLowSweptLondon) or (inNY and londonLowSweptNY) or (str.contains(sweptLongName, "Asia") and inLondon) or (str.contains(sweptLongName, "PDL") and (inLondon or inNY))
hierShort = (inLondon and asiaHighSweptLondon) or (inNY and londonHighSweptNY) or (str.contains(sweptShortName, "Asia") and inLondon) or (str.contains(sweptShortName, "PDH") and (inLondon or inNY))
hierOKL = not requireHier or hierLong
hierOKS = not requireHier or hierShort

// ═══════════════════════════ FILTERS ═══════════════════════════
body     = math.abs(close - open)
dispOK   = body >= dispMult * atrVal
bullCndl = close > open
bearCndl = close < open

biasOn    = biasMode == "EMA Filter"
longBias  = not biasOn or close > biasEma
shortBias = not biasOn or close < biasEma

longCorr  = (not useDXY or dxyC < dxyE) and (not useTNX or tnxC < tnxE)
shortCorr = (not useDXY or dxyC > dxyE) and (not useTNX or tnxC > tnxE)

stochLongRaw  = ta.lowest(k5, stochLB)  <= stochOS and ta.lowest(k15, stochLB)  <= stochOS
stochShortRaw = ta.highest(k5, stochLB) >= stochOB and ta.highest(k15, stochLB) >= stochOB
stochLongOK  = not useStoch or stochLongRaw
stochShortOK = not useStoch or stochShortRaw

ptjLongOK  = not usePTJ or close > sma200d
ptjShortOK = not usePTJ or close < sma200d
volAvg = ta.sma(volume, volLen)
volOK  = not useVolSpike or (not na(volume) and volume >= volMult * volAvg)
rangeUsed = adrD > 0 ? (cdH - cdL) / adrD : 0.0
adrSpent  = useADR and rangeUsed >= adrCap

// Confluence score (v3: +MSS quality, +session hierarchy, +round/EQ major)
scoreL = (biasOn and close > biasEma ? 1 : 0) + (useDXY and dxyC < dxyE ? 1 : 0) + (useStoch and stochLongRaw ? 1 : 0) + (inLondon or inNY or inSB ? 1 : 0) + (usePTJ and close > sma200d ? 1 : 0) + (sBull ? 1 : 0) + (adrSpent ? 1 : 0) + (useSessHier and hierLong ? 1 : 0) + (qualityBull ? 1 : 0)
scoreS = (biasOn and close < biasEma ? 1 : 0) + (useDXY and dxyC > dxyE ? 1 : 0) + (useStoch and stochShortRaw ? 1 : 0) + (inLondon or inNY or inSB ? 1 : 0) + (usePTJ and close < sma200d ? 1 : 0) + (sBear ? 1 : 0) + (adrSpent ? 1 : 0) + (useSessHier and hierShort ? 1 : 0) + (qualityBear ? 1 : 0)
gradeL = scoreL >= 6 ? "A" : scoreL >= 4 ? "B" : "C"
gradeS = scoreS >= 6 ? "A" : scoreS >= 4 ? "B" : "C"
gradeOKL = gradeMode == "All" or (gradeMode == "A + B" and gradeL != "C") or (gradeMode == "A only" and gradeL == "A")
gradeOKS = gradeMode == "All" or (gradeMode == "A + B" and gradeS != "C") or (gradeMode == "A only" and gradeS == "A")

hardF   = strictMode == "Hard filters"
gBiasL  = not hardF or longBias
gBiasS  = not hardF or shortBias
gCorrL  = not hardF or longCorr
gCorrS  = not hardF or shortCorr
gStochL = not hardF or stochLongOK
gStochS = not hardF or stochShortOK
gPtjL   = not hardF or ptjLongOK
gPtjS   = not hardF or ptjShortOK

allowRev  = tradeMode != "Continuation (trade acceptance)"
allowCont = tradeMode != "Reversal (fade sweep)"

// ═══════════════════════════ SIGNAL GUARDS ═══════════════════════════
var int signalsToday = 0
var int lastSigBar   = -100000
if newDay
    signalsToday := 0

atrBase       = ta.sma(atrVal, 200)
volChaos      = useStay and atrVal > volBandHi * atrBase
volDead       = useStay and atrVal < volBandLo * atrBase
macroConflict = useStay and useConflict and biasOn and useDXY and ((close > biasEma and dxyC > dxyE) or (close < biasEma and dxyC < dxyE))
friLate       = useStay and useFriClose and dayofweek(time, TZ_NY) == dayofweek.friday and not na(time(timeframe.period, friSess, TZ_NY))
capHit        = signalsToday >= maxSigDay
standAside    = volChaos or volDead or macroConflict or friLate or (useStay and capHit)
string stayWhy = volChaos ? "News-level volatility (Livermore: sit out)" : volDead ? "Dead market - nothing to win" : macroConflict ? "Bias vs DXY conflict = no conviction (Druckenmiller)" : friLate ? "Friday close-out - weekend risk" : capHit ? "Daily limit hit - stop (PTJ: do not overtrade)" : ""

canSignal = signalsToday < maxSigDay and (bar_index - lastSigBar) >= coolB and not standAside

// ═══════════════════════════ INSTITUTIONAL PENDING STATE ═══════════════════════════
// After a sweep, we do NOT fire immediately in PRO mode.
// We wait for: CISD (optional arm) → MSS (required) → OB/FVG retest entry.
var int   pendDir = 0          // +1 long, -1 short
var int   pendBars = 0
var float pendExtreme = na
var float pendMSSLevel = na    // swing that must be broken by body close
var bool  pendMSS = false
var bool  pendCISD = false
var string pendTag = ""
var string pendPool = ""
var float obTop = na
var float obBot = na
var float fvgTop = na
var float fvgBot = na
var int   waitEntry = 0
var int   zoneBorn = 0
var box   zoneBox = na
var float armSL = na

f_findBullOB() =>
    float top = na
    float bot = na
    for i = 1 to obLB
        if close[i] < open[i]
            top := high[i]
            bot := low[i]
            break
    [top, bot]

f_findBearOB() =>
    float top = na
    float bot = na
    for i = 1 to obLB
        if close[i] > open[i]
            top := high[i]
            bot := low[i]
            break
    [top, bot]

f_resetPend() =>
    pendDir := 0
    pendBars := 0
    pendExtreme := na
    pendMSSLevel := na
    pendMSS := false
    pendCISD := false
    pendTag := ""
    pendPool := ""
    obTop := na
    obBot := na
    fvgTop := na
    fvgBot := na
    waitEntry := 0
    zoneBorn := 0
    armSL := na
    if not na(zoneBox)
        box.delete(zoneBox)
        zoneBox := na

// Raw reversal setups (filters agree + sweep happened)
revLongSetup  = allowRev and anyBull and bullCndl and dispOK and gBiasL and gCorrL and sessionOK and gStochL and gPtjL and volOK and gradeOKL and hierOKL and (not hardF or qualityBull or minWickRatio <= 0)
revShortSetup = allowRev and anyBear and bearCndl and dispOK and gBiasS and gCorrS and sessionOK and gStochS and gPtjS and volOK and gradeOKS and hierOKS and (not hardF or qualityBear or minWickRatio <= 0)
contLongSig   = allowCont and contLong  and bullCndl and dispOK and gBiasL and gCorrL and sessionOK and gPtjL and volOK and gradeOKL and not adrSpent and canSignal
contShortSig  = allowCont and contShort and bearCndl and dispOK and gBiasS and gCorrS and sessionOK and gPtjS and volOK and gradeOKS and not adrSpent and canSignal

bool  fireLong  = false
bool  fireShort = false
float slLong    = na
float slShort   = na
string tagLong  = ""
string tagShort = ""
bool  mssLongMark = false
bool  mssShortMark = false
bool  cisdLongMark = false
bool  cisdShortMark = false

// ── Legacy reclaim entry ──
if revLongSetup and reclaimMode and canSignal
    fireLong := true
    slLong   := sweepRefLong - slBufATR * atrVal
    tagLong  := "Rev-Long " + gradeL
if revShortSetup and reclaimMode and canSignal
    fireShort := true
    slShort   := sweepRefShort + slBufATR * atrVal
    tagShort  := "Rev-Short " + gradeS

// ── Arm institutional pending on sweep (FVG or PRO modes) ──
if revLongSetup and fvgMode and pendDir == 0
    pendDir := 1
    pendBars := mssWinB
    pendExtreme := sweepRefLong
    // MSS level = last swing high before/at the SSL sweep (structure that defined the run)
    pendMSSLevel := na(swingHi) ? high[1] : swingHi
    pendMSS := not useMSS
    pendCISD := not useCISD
    pendTag := gradeL
    pendPool := sweptLongName
    armSL := sweepRefLong - slBufATR * atrVal
    if not proMode
        pendBars := fvgFB
    if proMode and useOB and pendMSS
        [ot, ob] = f_findBullOB()
        obTop := ot
        obBot := ob

if revShortSetup and fvgMode and pendDir == 0
    pendDir := -1
    pendBars := mssWinB
    pendExtreme := sweepRefShort
    pendMSSLevel := na(swingLo) ? low[1] : swingLo
    pendMSS := not useMSS
    pendCISD := not useCISD
    pendTag := gradeS
    pendPool := sweptShortName
    armSL := sweepRefShort + slBufATR * atrVal
    if not proMode
        pendBars := fvgFB
    if proMode and useOB and pendMSS
        [ot, ob] = f_findBearOB()
        obTop := ot
        obBot := ob

// Track worst extreme for double-sweep protection while pending
if pendDir == 1 and doubleSweep
    pendExtreme := math.min(nz(pendExtreme, low), low)
    armSL := pendExtreme - slBufATR * atrVal
if pendDir == -1 and doubleSweep
    pendExtreme := math.max(nz(pendExtreme, high), high)
    armSL := pendExtreme + slBufATR * atrVal

// CISD: first opposite-delivery close after sweep
if pendDir == 1 and useCISD and not pendCISD and close > open and close > open[1]
    pendCISD := true
    cisdLongMark := true
if pendDir == -1 and useCISD and not pendCISD and close < open and close < open[1]
    pendCISD := true
    cisdShortMark := true

// MSS: body close through the protected swing
if pendDir == 1 and useMSS and not pendMSS and not na(pendMSSLevel) and close > pendMSSLevel and close > open
    pendMSS := true
    mssLongMark := true
    if useOB
        [ot, ob] = f_findBullOB()
        obTop := ot
        obBot := ob
if pendDir == -1 and useMSS and not pendMSS and not na(pendMSSLevel) and close < pendMSSLevel and close < open
    pendMSS := true
    mssShortMark := true
    if useOB
        [ot, ob] = f_findBearOB()
        obTop := ot
        obBot := ob

// In classic FVG mode (non-PRO): form FVG during arm window without MSS gate
if pendDir != 0 and not proMode and waitEntry == 0
    if pendDir == 1
        if low > high[2]
            fvgBot := high[2]
            fvgTop := low
            waitEntry := fvgWB
            zoneBorn := bar_index
            if showFVGBox
                zoneBox := box.new(bar_index - 2, fvgTop, bar_index + fvgWB, fvgBot, border_color = color.green, bgcolor = color.new(color.green, 88))
            pendBars := 0
        else
            pendBars -= 1
            if pendBars <= 0
                f_resetPend()
    else
        if high < low[2]
            fvgTop := low[2]
            fvgBot := high
            waitEntry := fvgWB
            zoneBorn := bar_index
            if showFVGBox
                zoneBox := box.new(bar_index - 2, fvgTop, bar_index + fvgWB, fvgBot, border_color = color.red, bgcolor = color.new(color.red, 88))
            pendBars := 0
        else
            pendBars -= 1
            if pendBars <= 0
                f_resetPend()

// PRO mode: after MSS, build OB/FVG entry zone
if pendDir != 0 and proMode and pendMSS and waitEntry == 0
    bool zoneReady = false
    // Prefer FVG from displacement; fall back to Order Block
    if pendDir == 1
        if low > high[2]
            fvgBot := high[2]
            fvgTop := low
            zoneReady := true
        else if useOB and not na(obTop)
            fvgTop := obTop
            fvgBot := obBot
            zoneReady := true
    else
        if high < low[2]
            fvgTop := low[2]
            fvgBot := high
            zoneReady := true
        else if useOB and not na(obTop)
            fvgTop := obTop
            fvgBot := obBot
            zoneReady := true
    if zoneReady
        waitEntry := fvgWB
        zoneBorn := bar_index
        if showFVGBox
            color zc = pendDir == 1 ? color.green : color.red
            zoneBox := box.new(bar_index - 2, fvgTop, bar_index + fvgWB, fvgBot, border_color = zc, bgcolor = color.new(zc, 88), text = pendDir == 1 ? "BUY OB/FVG" : "SELL OB/FVG", text_color = zc)
        pendBars := 0
    else
        pendBars -= 1
        if pendBars <= 0
            f_resetPend()

// Expire pending if MSS never prints
if pendDir != 0 and proMode and not pendMSS and waitEntry == 0
    pendBars -= 1
    if pendBars <= 0
        f_resetPend()

// Retest entry into OB/FVG zone
if waitEntry > 0
    if pendDir == 1
        if close < armSL
            f_resetPend()
        else if bar_index > zoneBorn and low <= fvgTop and canSignal and (not proMode or pendMSS)
            fireLong := true
            slLong   := armSL
            tagLong  := (proMode ? "PRO-Long " : "FVG-Long ") + pendTag + (pendPool != "" ? " " + pendPool : "")
            f_resetPend()
        else
            waitEntry -= 1
            if waitEntry == 0
                f_resetPend()
    else if pendDir == -1
        if close > armSL
            f_resetPend()
        else if bar_index > zoneBorn and high >= fvgBot and canSignal and (not proMode or pendMSS)
            fireShort := true
            slShort   := armSL
            tagShort  := (proMode ? "PRO-Short " : "FVG-Short ") + pendTag + (pendPool != "" ? " " + pendPool : "")
            f_resetPend()
        else
            waitEntry -= 1
            if waitEntry == 0
                f_resetPend()

// Continuation still fires immediately (acceptance = real breakout)
if contLongSig and not fireLong
    fireLong := true
    slLong   := contLongBase - slBufATR * atrVal
    tagLong  := "Cont-Long " + gradeL
if contShortSig and not fireShort
    fireShort := true
    slShort   := contShortBase + slBufATR * atrVal
    tagShort  := "Cont-Short " + gradeS

// ═══════════════════════════ TARGET HELPERS ═══════════════════════════
f_bestAbove(array<LiqLevel> arr, float curBest) =>
    float best = curBest
    if array.size(arr) > 0
        for L in arr
            if L.enabled and L.kind == "high" and not na(L.price) and L.price > close
                best := na(best) ? L.price : math.min(best, L.price)
    best

f_bestBelow(array<LiqLevel> arr, float curBest) =>
    float best = curBest
    if array.size(arr) > 0
        for L in arr
            if L.enabled and L.kind == "low" and not na(L.price) and L.price < close
                best := na(best) ? L.price : math.max(best, L.price)
    best

f_oppLong() =>
    float best = f_bestAbove(staticLevels, na)
    best := f_bestAbove(dynLevels, best)
    best := f_bestAbove(roundLevels, best)
    best

f_oppShort() =>
    float best = f_bestBelow(staticLevels, na)
    best := f_bestBelow(dynLevels, best)
    best := f_bestBelow(roundLevels, best)
    best

// ═══════════════════════════ SIGNAL OUTPUT ═══════════════════════════
f_drawTrade(isLong, entryP, slP, tp1P, tp2P, qtyTxt, tag, timeTxt) =>

f_pinLevels(array<LiqLevel> arr, array<float> placed, float tol) =>
    for L in arr
        if not na(L.ln)
            line.set_extend(L.ln, extend.none)
            line.set_x2(L.ln, bar_index)
        if not na(L.lb)
            float yy = label.get_y(L.lb)
            int bump = 0
            if array.size(placed) > 0
                for p in placed
                    if math.abs(yy - p) <= tol
                        bump += 1
            label.set_x(L.lb, bar_index + pinOffset + bump * pinStagger)
            label.set_style(L.lb, label.style_label_left)
            array.push(placed, yy)
