// Pretty-print session input "0830-1100" / "0830-1100:23456" → "08:30-11:00"
f_sessPretty(string sess) =>
    int colon = str.pos(sess, ":")
    string body = not na(colon) ? str.substring(sess, 0, colon) : sess
    int dash = str.pos(body, "-")
    if na(dash)
        body
    else
        string a = str.substring(body, 0, dash)
        string b = str.substring(body, dash + 1, str.length(body))
        string fa = str.length(a) >= 4 ? str.substring(a, 0, 2) + ":" + str.substring(a, 2, 4) : a
        string fb = str.length(b) >= 4 ? str.substring(b, 0, 2) + ":" + str.substring(b, 2, 4) : b
        fa + "-" + fb

type LiqLevel
    string name

method refresh(LiqLevel L, float newPrice) =>
    L.price   := newPrice
    L.broke   := false
    L.extreme := na
    L.barsOut := 0
    L.armed   := true
    if not na(L.ln)
        line.delete(L.ln)
    if not na(L.lb)
        label.delete(L.lb)
    if L.enabled and not na(newPrice)
        string side = L.kind == "high" ? "BSL" : "SSL"
        L.pool := side
        // Keep previous full wording — tidy via pin stagger only
        // Plain English action tip on every level
        string tip = not useSimple ? " [" + side + "]" : L.kind == "high" ? " | up=SELL" : " | down=BUY"
        string txt = L.name + "  " + str.tostring(newPrice, format.mintick) + tip
        L.ln := line.new(bar_index, newPrice, bar_index + 1, newPrice, extend = extend.right, color = L.col, width = 2)
        L.lb := label.new(bar_index, newPrice, txt, style = label.style_label_right, color = f_lblBg(L.col), textcolor = f_lblTx(L.col), size = size.small)

method track(LiqLevel L, float minPen, int window) =>
    bool bearSweep = false
    bool bullSweep = false
    bool bullBreak = false
    bool bearBreak = false
    string sweptName = ""
    if L.enabled and L.armed and not na(L.price)
        if L.kind == "high"
            if high > L.price
                L.extreme := L.broke ? math.max(L.extreme, high) : high
                L.broke   := true
                if close > L.price
                    L.barsOut += 1
            if L.broke and close < L.price and (L.extreme - L.price) >= minPen and L.barsOut <= window
                bearSweep := true
                sweptName := L.name
                L.broke   := false
                L.armed   := reArm
            else if L.broke and L.barsOut > window and close > L.price
                bullBreak := true
                L.broke   := false
                L.armed   := false
        else
            if low < L.price
                L.extreme := L.broke ? math.min(L.extreme, low) : low
                L.broke   := true
                if close < L.price
                    L.barsOut += 1
            if L.broke and close > L.price and (L.price - L.extreme) >= minPen and L.barsOut <= window
                bullSweep := true
                sweptName := L.name
                L.broke   := false
                L.armed   := reArm
            else if L.broke and L.barsOut > window and close < L.price
                bearBreak := true
                L.broke   := false
                L.armed   := false
    [bearSweep, bullSweep, bullBreak, bearBreak, sweptName]

    string dailyBiasTxt = not useDailyBias ? "Off" : dailyBuyLean ? (useSimple ? "BUY ONLY today (discount vs open)" : "BUY day locked") : dailySellLean ? (useSimple ? "SELL ONLY today (premium vs open)" : "SELL day locked") : (useSimple ? "Mixed — no side yet" : "Mixed")
    string dailyOpenTxt = na(biasOpenRef) ? "-" : (belowOpen ? (useSimple ? "Below open = cheaper zone" : "Disc vs open") : aboveOpen ? (useSimple ? "Above open = richer zone" : "Prem vs open") : "At open")

" · NOW below = BUY side" : " · below") : close > midOpen ? (useSimple ? " · NOW above = SELL side" : " · above") : (useSimple ? " · at open" : " · at")))
    string dayDateTxt = na(dayOpenTime) ? (useSimple ? "no date yet" : "-") : (str.format_time(dayOpenTime, "dd MMM yyyy", TZ_NY) + " NY · " + str.format_time(dayOpenTime, "dd MMM HH:mm", sigTZ) + " Melb")
    string weekDateTxt = na(weekOpenTime) ? (useSimple ? "no date yet" : "-") : (str.format_time(weekOpenTime, "dd MMM yyyy", TZ_NY) + " NY · " + str.format_time(weekOpenTime, "dd MMM HH:mm", sigTZ) + " Melb")
    string dayMonitorTxt = not useOpens ? "Off" : na(dayOpen) ? (useSimple ? "Waiting for NY day open" : "n/a") : (str.tostring(dayOpen, format.mintick) + " · " + dayDateTxt + (close < dayOpen ? (useSimple ? " · below · BUY can cash out here" : " · below") : close > dayOpen ? (useSimple ? " · above · SELL can cash out here" : " · above") : (useSimple ? " · at open" : " · at")))
    string weekMonitorTxt = not useOpens ? "Off" : na(weekOpen) ? (useSimple ? "Waiting for NY week open" : "n/a") : (str.tostring(weekOpen, format.mintick) + " · " + weekDateTxt + (close < weekOpen ? (useSimple ? " · below" : " · below") : close > weekOpen ? (useSimple ?     scoreL = (inLondon or inNY or inSB ? 1 : 0) + (extQualityL ? 1 : 0) + (nestedLong ? 1 : 0) + (premDiscLongOK ? 1 : 0) + (smtBullRaw ? 2 : 0) + (qualityBull ? 1 : 0) + (useSessHier and hierLong ? 1 : 0) + (nearOpen or nearBell or nearNDOG or nearNWOG ? 1 : 0) + (adrSpent ? 1 : 0) + (biasUp4h ? 1 : 0) + (useStoch and stochLongRaw ? 1 : 0) + (not na(drawBSL) ? 1 : 0) + (inHtfBullPOI ? 1 : 0) + (inBullIFVG ? 1 : 0) + (nearBPR ? 1 : 0) + (amdPhase == "DIST" ? 1 : 0) + (timeConf and useTimeConf ? 1 : 0) + (dailyBuyLean ? 2 : 0) + (judasBuyPath ? 2 : 0) + (sbPlaybook and inSB and v_sbLoSwept ? 3 : sbPlaybook and inSB ? 1 : 0) + (dispStrong and bullCndl and bullDispFVG ? 2 : dispStrong and bullCndl ? 1 : 0) + (smtEsBull ? 1 : 0) + (useIPDA and not na(lvlIP20L.price) and math.abs(close - lvlIP20L.price) <= atrVal ? 1 : 0)

    gradeL = scoreL >= 10 ? "A+" : scoreL >= 7 ? "A" : scoreL >= 5 ? "B" : "C"

    gradeOKL = gradeMode == "All" or (gradeMode == "A + B" and gradeL != "C") or (gradeMode == "A only" and (gradeL == "A" or gradeL == "A+"))
    gradeOKS = gradeMode == "All" or (gradeMode == "A + B" and gradeS != "C") or (gradeMode == "A only" and (gradeS == "A" or gradeS == "A+"))

    // Pure Liquidity mode: EMA / Stoch / PTJ never veto — liquidity gates do
    hardF   = strictMode == "Hard filters"
    softInd = pureLiq  // indirect filters become soft
    gBiasL  = softInd or not hardF or longBias
    gBiasS  = softInd or not hardF or shortBias
    gCorrL  = softInd or not hardF or longCorr
    gCorrS  = softInd or not hardF or shortCorr
    gStochL = softInd or not hardF or stochLongOK
    gStochS = softInd or not hardF or stochShortOK
    gPtjL   = softInd or not hardF or ptjLongOK
    gPtjS   = softInd or not hardF or ptjShortOK

    allowRev  = tradeMode != "Continuation (trade acceptance)"
    allowCont = tradeMode != "Reversal (fade sweep)"

    // ═══════════════════════════ SIGNAL GUARDS ═══════════════════════════
    if newDay
        v_signalsToday := 0

    atrBase       = ta.sma(atrVal, 200)
    volChaos      = useStay and atrVal > volBandHi * atrBase
    volDead       = useStay and atrVal < volBandLo * atrBase
    macroConflict = useStay and useConflict and biasOn and useDXY and ((close > biasEma and dxyC > dxyE) or (close < biasEma and dxyC < dxyE))
    friLate       = useStay and useFriClose and dayofweek(time, TZ_NY) == dayofweek.friday and not na(time(timeframe.period, friSess, TZ_NY))
    capHit        = v_signalsToday >= maxSigDay
    standAside    = volChaos or volDead or macroConflict or friLate or (useStay and capHit) or inNews or inDeadHrs
    string stayWhy = inNews ? (useSimple ? "News time — do not trade" : "News blackout (CPI/NFP/FOMC window)") : inDeadHrs ? (useSimple ? (inLonClose ? "London close — stand aside" : "Lunch chop — stand aside") : (inLonClose ? "London close dead hour" : "NY lunch dead hour")) : volChaos ? (useSimple ? "Market too wild — sit out" : "News-level volatility (Livermore: sit out)") : volDead ? (useSimple ? "Market too quiet — sit out" : "Dead market - nothing to win") : macroConflict ? (useSimple ? "Dollar conflict — no clear side" : "Bias vs DXY conflict = no conviction (Druckenmiller)") : friLate ? (useSimple ? "Friday late — weekend risk" : "Friday close-out - weekend risk") : capHit ? (useSimple ? "Daily trade limit reached" : "Daily limit hit - stop (PTJ: do not overtrade)") : ""

    canSignal = v_signalsToday < maxSigDay and (bar_index - lastSigBar) >= coolB and not standAside and sbHourOK and kzSlotOK
    PlaySnap o = PlaySnap.new()
    o.lockedBias := v_lockedBias
    o.biasLockDay := v_biasLockDay
    o.sbLoSwept := v_sbLoSwept

            break
    [top, bot]

f_runPendingAndFire() =>
    int v_pendDir = pendDir
    int v_pendBars = pendBars
    float v_pendExtreme = pendExtreme
    float v_pendMSSLevel = pendMSSLevel
    bool v_pendMSS = pendMSS
    bool v_pendCISD = pendCISD
    string v_pendTag = pendTag
    string v_pendPool = pendPool
    float v_obTop = obTop
    float v_obBot = obBot
    float v_fvgTop = fvgTop
    float v_fvgBot = fvgBot
    float v_ceLevel = ceLevel
    float v_oteHi = oteHi
    float v_oteLo = oteLo
    float v_ote62 = ote62
    float v_ote79 = ote79
    float v_brkTop = brkTop
    float v_brkBot = brkBot
    int v_waitEntry = waitEntry
    int v_zoneBorn = zoneBorn
    box v_zoneBox = zoneBox
    float v_armSL = armSL
    // Flag to clear pending state in global scope (Pine forbids := on globals inside functions)
    bool clearPend = false

    // Raw reversal setups — liquidity/PA + elite gates + v6.1 complete playbook
    revLongSetup  = pb.allowRev and anyBull and bullCndl and dispOK and pb.gBiasL and pb.gCorrL and sessionOK and pb.gStochL and pb.gPtjL and pb.volOK and pb.gradeOKL and hierOKL and nestOKL and premDiscLongOK and smtLongOK and drawLongOK and poiLongOK and amdLongOK and aPlusOKL and pb.dailyLongOK and pb.judasLongOK and pb.dispLongOK and pb.sbModelLongOK and (not pb.hardF or qualityBull or minWickRatio <= 0)
    revShortSetup = pb.allowRev and anyBear and bearCndl and dispOK and pb.gBiasS and pb.gCorrS and sessionOK and pb.gStochS and pb.gPtjS and pb.volOK and pb.gradeOKS and hierOKS and nestOKS and premDiscShortOK and smtShortOK and drawShortOK and poiShortOK and amdShortOK and aPlusOKS and pb.dailyShortOK and pb.judasShortOK and pb.dispShortOK and pb.sbModelShortOK and (not pb.hardF or qualityBear or minWickRatio <= 0)
    contLongSig   = pb.allowCont and contLong  and bullCndl and dispOK and pb.gBiasL and pb.gCorrL and sessionOK and pb.gPtjL and pb.volOK and pb.gradeOKL and premDiscLongOK and not pb.adrSpent and pb.dailyLongOK and pb.dispLongOK and pb.sbModelLongOK and pb.canSignal
    contShortSig  = pb.allowCont and contShort and bearCndl and dispOK and pb.gBiasS and pb.gCorrS and sessionOK and pb.gPtjS and pb.volOK and pb.gradeOKS and premDiscShortOK and not pb.adrSpent and pb.dailyShortOK and pb.dispShortOK and pb.sbModelShortOK and pb.canSignal

    bool  v_fireLong  = false
    bool  v_fireShort = false
    float v_slLong    = na
    float v_slShort   = na
    string v_tagLong  = ""
    string v_tagShort = ""
    bool  v_mssLongMark = false
    bool  v_mssShortMark = false
    bool  v_cisdLongMark = false
    bool  v_cisdShortMark = false

    // ── Legacy reclaim entry ──
    if revLongSetup and reclaimMode and pb.canSignal
        v_fireLong := true
        v_slLong   := sweepRefLong - slBufATR * atrVal
        v_tagLong  := "Rev-Long " + pb.gradeL
    if revShortSetup and reclaimMode and pb.canSignal
        v_fireShort := true
        v_slShort   := sweepRefShort + slBufATR * atrVal
        v_tagShort  := "Rev-Short " + pb.gradeS

    // ── Arm institutional pending on sweep (FVG or PRO modes) ──
    if revLongSetup and fvgMode and v_pendDir == 0
        v_pendDir := 1
        v_pendBars := mssWinB
        v_pendExtreme := sweepRefLong
        // MSS level = last swing high before/at the SSL sweep (structure that defined the run)
        v_pendMSSLevel := na(swingHi) ? high[1] : swingHi
        v_pendMSS := not useMSS
        v_pendCISD := not useCISD
        v_pendTag := pb.gradeL
        v_pendPool := sweptLongName
        v_armSL := sweepRefLong - slBufATR * atrVal
        if not proMode
            v_pendBars := fvgFB
        if proMode and useOB and v_pendMSS
            [ot, ob] = f_findBullOB()
            v_obTop := ot
            v_obBot := ob

    if revShortSetup and fvgMode and v_pendDir == 0
        v_pendDir := -1
        v_pendBars := mssWinB
        v_pendExtreme := sweepRefShort
        v_pendMSSLevel := na(swingLo) ? low[1] : swingLo
        v_pendMSS := not useMSS
        v_pendCISD := not useCISD
        v_pendTag := pb.gradeS
        v_pendPool := sweptShortName
        v_armSL := sweepRefShort + slBufATR * atrVal
        if not proMode
            v_pendBars := fvgFB
        if proMode and useOB and v_pendMSS
            [ot, ob] = f_findBearOB()
            v_obTop := ot
            v_obBot := ob

    // Track worst extreme for double-sweep protection while pending
    if v_pendDir == 1 and doubleSweep
        v_pendExtreme := math.min(nz(v_pendExtreme, low), low)
        v_armSL := v_pendExtreme - slBufATR * atrVal
    if v_pendDir == -1 and doubleSweep
        v_pendExtreme := math.max(nz(v_pendExtreme, high), high)
        v_armSL := v_pendExtreme + slBufATR * atrVal

    // CISD: first opposite-delivery close after sweep
    if v_pendDir == 1 and useCISD and not v_pendCISD and close > open and close > open[1]
        v_pendCISD := true
        v_cisdLongMark := true
    if v_pendDir == -1 and useCISD and not v_pendCISD and close < open and close < open[1]
        v_pendCISD := true
        v_cisdShortMark := true

    // MSS: body close through the protected swing
    if v_pendDir == 1 and useMSS and not v_pendMSS and not na(v_pendMSSLevel) and close > v_pendMSSLevel and close > open
        v_pendMSS := true
        v_mssLongMark := true
        if useOB
            [ot, ob] = f_findBullOB()
            v_obTop := ot
            v_obBot := ob
        // OTE of bullish displacement: from sweep extreme (low) up to this MSS high
        v_oteLo := v_pendExtreme
        v_oteHi := high
        float rngL = v_oteHi - v_oteLo
        if rngL > 0
            v_ote62 := v_oteHi - 0.62 * rngL
            v_ote79 := v_oteHi - 0.79 * rngL
        // Breaker = broken swing shelf
        if useBreaker and not na(v_pendMSSLevel)
            v_brkTop := v_pendMSSLevel + 0.15 * atrVal
            v_brkBot := v_pendMSSLevel - 0.15 * atrVal
    if v_pendDir == -1 and useMSS and not v_pendMSS and not na(v_pendMSSLevel) and close < v_pendMSSLevel and close < open
        v_pendMSS := true
        v_mssShortMark := true
        if useOB
            [ot, ob] = f_findBearOB()
            v_obTop := ot
            v_obBot := ob
        v_oteHi := v_pendExtreme
        v_oteLo := low
        float rngS = v_oteHi - v_oteLo
        if rngS > 0
            v_ote62 := v_oteLo + 0.62 * rngS
            v_ote79 := v_oteLo + 0.79 * rngS
        if useBreaker and not na(v_pendMSSLevel)
            v_brkTop := v_pendMSSLevel + 0.15 * atrVal
            v_brkBot := v_pendMSSLevel - 0.15 * atrVal

    // In classic FVG mode (non-PRO): form FVG during arm window without MSS gate
    if v_pendDir != 0 and not proMode and v_waitEntry == 0
        if v_pendDir == 1
            if low > high[2]
                v_fvgBot := high[2]
                v_fvgTop := low
                v_waitEntry := fvgWB
                v_zoneBorn := bar_index
                if showFVGBox
                    v_zoneBox := box.new(bar_index - 2, v_fvgTop, bar_index + fvgWB, v_fvgBot, border_color = color.green, bgcolor = color.new(color.green, 88))
                v_pendBars := 0
            else
                v_pendBars -= 1
                if v_pendBars <= 0
                    clearPend := true
        else
            if high < low[2]
                v_fvgTop := low[2]
                v_fvgBot := high
                v_waitEntry := fvgWB
                v_zoneBorn := bar_index
                if showFVGBox
                    v_zoneBox := box.new(bar_index - 2, v_fvgTop, bar_index + fvgWB, v_fvgBot, border_color = color.red, bgcolor = color.new(color.red, 88))
                v_pendBars := 0
            else
                v_pendBars -= 1
                if v_pendBars <= 0
                    clearPend := true

    // PRO mode: after MSS, build FVG / OB / Breaker entry zone (+ CE)
    if v_pendDir != 0 and proMode and v_pendMSS and v_waitEntry == 0
        bool zoneReady = false
        if v_pendDir == 1
            if low > high[2]
                v_fvgBot := high[2]
                v_fvgTop := low
                zoneReady := true
            else if useOB and not na(v_obTop)
                v_fvgTop := v_obTop
                v_fvgBot := v_obBot
                zoneReady := true
            else if useBreaker and not na(v_brkTop)
                v_fvgTop := v_brkTop
                v_fvgBot := v_brkBot
                zoneReady := true
        else
            if high < low[2]
                v_fvgTop := low[2]
                v_fvgBot := high
                zoneReady := true
            else if useOB and not na(v_obTop)
                v_fvgTop := v_obTop
                v_fvgBot := v_obBot
                zoneReady := true
            else if useBreaker and not na(v_brkTop)
                v_fvgTop := v_brkTop
                v_fvgBot := v_brkBot
                zoneReady := true
        if zoneReady
            v_ceLevel := (v_fvgTop + v_fvgBot) / 2.0
            v_waitEntry := fvgWB
            v_zoneBorn := bar_index
            bool uniZone = useUnicorn and not na(v_brkTop) and not na(v_brkBot) and v_fvgBot <= v_brkTop and v_fvgTop >= v_brkBot
            if showFVGBox
                if v_pendDir == 1
                    v_zoneBox := box.new(bar_index - 2, v_fvgTop, bar_index + fvgWB, v_fvgBot, border_color = color.green, bgcolor = color.new(color.green, 88), text = useSimple ? (uniZone ? "BUY ZONE (best)" : "BUY ZONE — enter on dip") : (uniZone ? "BUY UNICORN" : "BUY CE/OB/FVG"), text_color = color.green)
                else
                    v_zoneBox := box.new(bar_index - 2, v_fvgTop, bar_index + fvgWB, v_fvgBot, border_color = color.red, bgcolor = color.new(color.red, 88), text = useSimple ? (uniZone ? "SELL ZONE (best)" : "SELL ZONE — enter on rise") : (uniZone ? "SELL UNICORN" : "SELL CE/OB/FVG"), text_color = color.red)
            v_pendBars := 0
        else
            v_pendBars -= 1
            if v_pendBars <= 0
                clearPend := true

    // Expire pending if MSS never prints
    if v_pendDir != 0 and proMode and not v_pendMSS and v_waitEntry == 0
        v_pendBars -= 1
        if v_pendBars <= 0
            clearPend := true

    // Retest entry: CE (50%) preferred, else zone edge; OTE filter when armed
    bool oteLongOK = not useOTE or na(v_ote62) or na(v_ote79) or ((low <= v_ote62 and high >= v_ote79) or (close <= v_ote62 and close >= v_ote79))
    bool oteShortOK = not useOTE or na(v_ote62) or na(v_ote79) or ((high >= v_ote62 and low <= v_ote79) or (close >= v_ote62 and close <= v_ote79))

    if v_waitEntry > 0
        if v_pendDir == 1
            float touchL = useCE and not na(v_ceLevel) ? v_ceLevel : v_fvgTop
            if close < v_armSL
                clearPend := true
            else if bar_index > v_zoneBorn and low <= touchL and oteLongOK and pb.canSignal and (not proMode or v_pendMSS)
                v_fireLong := true
                v_slLong   := v_armSL
                bool uniL = useUnicorn and not na(v_brkTop) and not na(v_brkBot) and not na(v_fvgTop) and v_fvgBot <= v_brkTop and v_fvgTop >= v_brkBot
                v_tagLong  := (proMode ? "PRO-Long " : "FVG-Long ") + v_pendTag + (v_pendPool != "" ? " " + v_pendPool : "") + (smtBullRaw ? " SMT" : "") + (uniL ? " UNI" : "") + (inBullIFVG ? " IFVG" : "") + (aPlusLong ? " A+" : "")
                clearPend := true
            else
                v_waitEntry -= 1
                if v_waitEntry == 0
                    clearPend := true
        else if v_pendDir == -1
            float touchS = useCE and not na(v_ceLevel) ? v_ceLevel : v_fvgBot
            if close > v_armSL
                clearPend := true
            else if bar_index > v_zoneBorn and high >= touchS and oteShortOK and pb.canSignal and (not proMode or v_pendMSS)
                v_fireShort := true
                v_slShort   := v_armSL
                bool uniS = useUnicorn and not na(v_brkTop) and not na(v_brkBot) and not na(v_fvgTop) and v_fvgBot <= v_brkTop and v_fvgTop >= v_brkBot
                v_tagShort  := (proMode ? "PRO-Short " : "FVG-Short ") + v_pendTag + (v_pendPool != "" ? " " + v_pendPool : "") + (smtBearRaw ? " SMT" : "") + (uniS ? " UNI" : "") + (inBearIFVG ? " IFVG" : "") + (aPlusShort ? " A+" : "")
                clearPend := true
            else
                v_waitEntry -= 1
                if v_waitEntry == 0
                    clearPend := true

    // Apply pending reset in global scope (required by Pine CE10088)
    if clearPend
        v_pendDir := 0
        v_pendBars := 0
        v_pendExtreme := na
        v_pendMSSLevel := na
        v_pendMSS := false
        v_pendCISD := false
        v_pendTag := ""
        v_pendPool := ""
        v_obTop := na
        v_obBot := na
        v_fvgTop := na
        v_fvgBot := na
        v_ceLevel := na
        v_oteHi := na
        v_oteLo := na
        v_ote62 := na
        v_ote79 := na
        v_brkTop := na
        v_brkBot := na
        v_waitEntry := 0
        v_zoneBorn := 0
        v_armSL := na
        if not na(v_zoneBox)
            box.delete(v_zoneBox)
            v_zoneBox := na

    // Continuation still fires immediately (acceptance = real breakout)
    if contLongSig and not v_fireLong
        v_fireLong := true
        v_slLong   := contLongBase - slBufATR * atrVal
        v_tagLong  := "Cont-Long " + pb.gradeL
    if contShortSig and not v_fireShort
        v_fireShort := true
        v_slShort   := contShortBase + slBufATR * atrVal
        v_tagShort  := "Cont-Short " + pb.gradeS

    // TESTER fallback: session + liquidity sweep reclaim (produces Strategy Tester trades)
    if testerLoose and pb.canSignal and sessionOK and strategy.position_size == 0
        if not v_fireLong and not v_fireShort and anyBull and bullCndl and close > open and (not usePremDisc or inDiscount or not effReqDaily)
            v_fireLong := true
            v_slLong := (na(v_armSL) ? low : math.min(low, v_armSL)) - slBufATR * atrVal
            v_tagLong := "Tester BUY " + (sweptLongName != "" ? sweptLongName : "sweep")
        else if not v_fireLong and not v_fireShort and anyBear and bearCndl and close < open and (not usePremDisc or inPremium or not effReqDaily)
            v_fireShort := true
            v_slShort := (na(v_armSL) ? high : math.max(high, v_armSL)) + slBufATR * atrVal
            v_tagShort := "Tester SELL " + (sweptShortName != "" ? sweptShortName : "sweep")
    PendSnap o = PendSnap.new()

f_bestAbove(array<LiqLevel> arr, float curBest) =>
    float best = curBest
    if array.size(arr) > 0
        for L in arr
            if L.enabled and L.kind == "high" and not na(L.price) and L.price > close
                best := na(best) ? L.price : math.min(best, L.price)
    best

f_bestBelow(array<LiqLevel> arr, float curBest) =>
    float best = curBest
    if array.size(arr) > 0
        for L in arr
            if L.enabled and L.kind == "low" and not na(L.price) and L.price < close
                best := na(best) ? L.price : math.max(best, L.price)
    best

f_oppLong() =>
    // Prefer HTF draw BSL when available
    float best = useHTFDraw and not na(drawBSL) ? drawBSL : na
    best := f_bestAbove(staticLevels, best)
    best := f_bestAbove(dynLevels, best)
    best := f_bestAbove(roundLevels, best)
    if useOpens
        if not na(dayOpen) and dayOpen > close

// Clean RR cards — TV long/short look, almost no text (table holds details)
var array<box>   rrBoxes = array.new_box()
var array<line>  rrLines = array.new_line()
var array<label> rrLabs  = array.new_label()

f_rrPrune() =>
    int maxB = rrKeepN * 2
    int maxL = rrKeepN * 2
    int maxLb = rrKeepN
    while array.size(rrBoxes) > maxB
        box.delete(array.shift(rrBoxes))
    while array.size(rrLines) > maxL
        line.delete(array.shift(rrLines))
    while array.size(rrLabs) > maxLb
        label.delete(array.shift(rrLabs))

f_drawTrade(isLong, entryP, slP, tp1P, tp2P, qtyTxt, tag, timeTxt) =>
    int x1 = bar_index
    int x2 = bar_index + rrBoxBars
    int xMid = int(math.round((x1 + x2) * 0.5))
    // Compact target = TP1 (keeps card height sane)
    float tpShow = not na(tp1P) ? tp1P : tp2P
    float risk = math.abs(entryP - slP)
    float reward = math.abs(tpShow - entryP)
    float rr = risk > 0 ? reward / risk : 0.0
    // TV-like soft fills
    color riskBg = color.new(color.rgb(239, 83, 80), 85)
    color riskBd = color.new(color.rgb(239, 83, 80), 55)
    color winBg  = color.new(color.rgb(38, 166, 154), 85)
    color winBd  = color.new(color.rgb(38, 166, 154), 55)

    if showRRBoxes and risk > 0 and reward > 0
        box bRisk = na
        box bWin = na
        if isLong
            // LONG card: profit ABOVE, risk BELOW (TV Long Position)
            bWin  := box.new(x1, tpShow, x2, entryP, bgcolor = winBg, border_color = winBd, border_width = 1)
            bRisk := box.new(x1, entryP, x2, slP, bgcolor = riskBg, border_color = riskBd, border_width = 1)
            // Teal stem under risk (TV cue)
            line stem = line.new(xMid, entryP, xMid, slP - risk * 0.25, color = color.new(color.rgb(38, 166, 154), 30), width = 2)
            array.push(rrLines, stem)
        else
            // SHORT card: risk ABOVE, profit BELOW (TV Short Position)
            bRisk := box.new(x1, slP, x2, entryP, bgcolor = riskBg, border_color = riskBd, border_width = 1)
            bWin  := box.new(x1, entryP, x2, tpShow, bgcolor = winBg, border_color = winBd, border_width = 1)
            line stem = line.new(xMid, slP, xMid, tpShow, color = color.new(color.gray, 50), width = 1, style = line.style_dashed)
            array.push(rrLines, stem)
        array.push(rrBoxes, bWin)
        array.push(rrBoxes, bRisk)
        // Entry seam only
        line eLn = line.new(x1, entryP, x2, entryP, color = color.new(color.rgb(55, 71, 79), 25), width = 1)
        array.push(rrLines, eLn)
        // ONE mid pill — no Target/Stop/Amount/Qty clutter (those live in table)
        string pill = (isLong ? "BUY " : "SELL ") + str.tostring(rr, "#.#") + "R"
        label lb = label.new(x2, entryP, pill, style = label.style_label_left, color = isLong ? color.new(color.rgb(38, 166, 154), 15) : color.new(color.rgb(239, 83, 80), 15), textcolor = color.white, size = size.tiny)
        array.push(rrLabs, lb)
        f_rrPrune()

    if showTradeLines or not showRRBoxes
        line.new(bar_index, entryP, bar_index + lineExt, entryP, color = color.yellow, width = 2)
        label.new(bar_index + lineExt, entryP, (isLong ? "BUY " : "SELL ") + tag + "  Entry: " + str.tostring(entryP, format.mintick) + "  Size: " + qtyTxt + "  " + timeTxt, color = color.yellow, textcolor = color.black, style = label.style_label_left, size = size.small)
        line.new(bar_index, slP, bar_index + lineExt, slP, color = color.red, width = 2)
        label.new(bar_index + lineExt, slP, "SL: " + str.tostring(slP, format.mintick), color = color.red, textcolor = color.white, style = label.style_label_left, size = size.small)
        line.new(bar_index, tp1P, bar_index + lineExt, tp1P, color = color.green, width = 1)
        label.new(bar_index + lineExt, tp1P, "TP1: " + str.tostring(tp1P, format.mintick) + (useSimple ? " - take half, move SL to entry" : " (half off, SL to BE)"), color = color.green, textcolor = color.white, style = label.style_label_left, size = size.small)
        line.new(bar_index, tp2P, bar_index + lineExt, tp2P, color = color.teal, width = 2)
        label.new(bar_index + lineExt, tp2P, "TP2: " + str.tostring(tp2P, format.mintick) + (useSimple ? " - final cash-out" : ""), color = color.teal, textcolor = color.white, style = label.style_label_left, size = size.small)

// Rebuild a level's short base text (avoids appending forever)
f_levelBaseTxt(LiqLevel L) =>
    bool isRnd = str.startswith(L.name, "RND")
    bool nameHasPx = str.contains(L.name, str.tostring(math.round(L.price)))

    useSimple ? (isRnd ? (L.name + (L.kind == "high" ? " · sell if fake↑" : " · buy if fake↓")) : (L.name + pxBit + (L.kind == "high" ? " · sell if fake↑" : " · buy if fake↓"))) : (L.name + " [" + L.pool + "] " + str.tostring(L.price, format.mintick))

// Try merge cash-out tag into the nearest existing label on that shelf. Returns true if merged.
f_mergeCashTag(float cashPx, bool isBuy) =>
    bool merged = false
    string tag = useSimple ? (isBuy ? " · +BUY exit" : " · +SELL exit") : (isBuy ? " · BSL" : " · SSL")
    float bestDist = 1e12
    label bestLb = na
    string bestBase = ""
    if useOpens
        if not na(dayOpenLb) and f_nearPx(cashPx, dayOpen)
            float d = math.abs(cashPx - dayOpen)
            if d < bestDist
                bestDist := d
                bestLb := dayOpenLb
                bestBase := f_samePrint(dayOpen, midOpen) ? f_dayMidTxt() : f_dayOpenTxt()
        if not na(weekOpenLb) and f_nearPx(cashPx, weekOpen)
            float d = math.abs(cashPx - weekOpen)
            if d < bestDist
                bestDist := d
                bestLb := weekOpenLb
                bestBase := f_weekOpenTxt()
        if not na(midOpenLb) and f_nearPx(cashPx, midOpen)
            float d = math.abs(cashPx - midOpen)
            if d < bestDist
                bestDist := d
                bestLb := midOpenLb
                bestBase := f_samePrint(dayOpen, midOpen) ? f_dayMidTxt() : f_midOpenTxt()
    if array.size(staticLevels) > 0
        for L in staticLevels
            if not na(L.lb) and not na(L.price) and f_nearPx(cashPx, L.price)
                float d = math.abs(cashPx - L.price)
                if d < bestDist
                    bestDist := d
                    bestLb := L.lb
                    bestBase := f_levelBaseTxt(L)
    if array.size(dynLevels) > 0
        for L in dynLevels
            if not na(L.lb) and not na(L.price) and f_nearPx(cashPx, L.price)
                float d = math.abs(cashPx - L.price)
                if d < bestDist
                    bestDist := d
                    bestLb := L.lb
                    bestBase := f_levelBaseTxt(L)
    if array.size(roundLevels) > 0
        for L in roundLevels
            if not na(L.lb) and not na(L.price) and f_nearPx(cashPx, L.price)
                float d = math.abs(cashPx - L.price)
                if d < bestDist
                    bestDist := d
                    bestLb := L.lb
                    bestBase := f_levelBaseTxt(L)
    if not na(bestLb)
        label.set_text(bestLb, bestBase + tag)
        merged := true
    merged

// ═══════════════════════════ SMART LABEL PINNING (anti-overlap UX) ═══════════════════════════
// Sort-free cluster bump: each new label near an existing one gets a higher
// horizontal stagger + small vertical ATR nudge so text never stacks.
var array<float> pinYs = array.new_float()
var array<int>   pinBumps = array.new_int()

f_pinOne(label lb, line ln, float yy) =>
    if not na(ln)
        line.set_extend(ln, extend.none)
        line.set_x2(ln, bar_index)
    if not na(lb) and not na(yy)
        float tol = atrVal * collideTolATR
        int bump = 0
        // Ladder every label that sits near another — keep ALL text, just fan them out
        if array.size(pinYs) > 0
            for i = 0 to array.size(pinYs) - 1
                if math.abs(yy - array.get(pinYs, i)) <= tol
                    bump := math.max(bump, array.get(pinBumps, i) + 1)
        bump := math.min(bump, maxBump)
        // Alternate vertical direction each step so a dense cluster fans up AND down
        float yDir = bump % 2 == 0 ? 1.0 : -1.0
        float yShift = math.ceil(bump / 2.0) * atrVal * yNudgeATR * yDir
        label.set_x(lb, bar_index + pinOffset + bump * pinStagger)
        label.set_y(lb, yy + yShift)
        label.set_style(lb, label.style_label_left)
        array.push(pinYs, yy)
        array.push(pinBumps, bump)

f_collectPin(array<LiqLevel> arr, array<label> lbs, array<line> lns, array<float> ys) =>
    if array.size(arr) > 0
        for L in arr
            if not na(L.lb) and not na(L.price)
                array.push(lbs, L.lb)
                array.push(lns, L.ln)
                array.push(ys, L.price)

// Re-apply readable colors every bar (fixes already-drawn dark labels like NY midnight)
f_restyleLevels(array<LiqLevel> arr) =>
    if array.size(arr) > 0
        for L in arr
            f_styleLb(L.lb, L.col)

f_runPinLabels() =>
    if pinLabels and barstate.islast
        array.clear(pinYs)
        array.clear(pinBumps)
        // Gather EVERY label (keep all), sort by price, then fan out — tidy only
        array<label> allLbs = array.new_label()
        array<line>  allLns = array.new_line()
        array<float> allYs  = array.new_float()
        f_collectPin(staticLevels, allLbs, allLns, allYs)
        f_collectPin(dynLevels, allLbs, allLns, allYs)
        f_collectPin(roundLevels, allLbs, allLns, allYs)
        if useOpens
            if not na(dayOpenLb) and not na(dayOpen)
                array.push(allLbs, dayOpenLb)
                array.push(allLns, dayOpenLn)
                array.push(allYs, dayOpen)
            if not na(weekOpenLb) and not na(weekOpen)
                array.push(allLbs, weekOpenLb)
                array.push(allLns, weekOpenLn)
                array.push(allYs, weekOpen)
            // Skip midnight label when merged into day open
            if not na(midOpenLb) and not na(midOpen) and not f_samePrint(dayOpen, midOpen)
                array.push(allLbs, midOpenLb)
                array.push(allLns, midOpenLn)
                array.push(allYs, midOpen)
        if useXBells
            if not na(lonOpenLb) and not na(lonOpenPx)
                array.push(allLbs, lonOpenLb)
                array.push(allLns, lonOpenLn)
                array.push(allYs, lonOpenPx)
            if not na(lonCloseLb) and not na(lonClosePx)
                array.push(allLbs, lonCloseLb)
                array.push(allLns, lonCloseLn)
                array.push(allYs, lonClosePx)
            if not na(usOpenLb) and not na(usOpenPx)
                array.push(allLbs, usOpenLb)
                array.push(allLns, usOpenLn)
                array.push(allYs, usOpenPx)
            if not na(nyseCloseLb) and not na(nyseClosePx)
                array.push(allLbs, nyseCloseLb)
                array.push(allLns, nyseCloseLn)
                array.push(allYs, nyseClosePx)
            if not na(asiaOpenLb) and not na(asiaOpenPx)
                array.push(allLbs, asiaOpenLb)
                array.push(allLns, asiaOpenLn)
                array.push(allYs, asiaOpenPx)
        // Cash-out labels (only if not merged into an open) — pin with the rest
        if not na(drawUpLb) and not na(drawBSL)
            array.push(allLbs, drawUpLb)
            array.push(allLns, drawUpLn)
            array.push(allYs, drawBSL)
        if not na(drawDnLb) and not na(drawSSL)
            array.push(allLbs, drawDnLb)
            array.push(allLns, drawDnLn)
            array.push(allYs, drawSSL)
        // Bubble-sort by price so nearby levels get sequential stagger steps
        int n = array.size(allYs)
        if n > 1
            for i = 0 to n - 2
                for j = 0 to n - 2 - i
                    if array.get(allYs, j) > array.get(allYs, j + 1)
                        float ty = array.get(allYs, j)
                        label tlb = array.get(allLbs, j)
                        line tln = array.get(allLns, j)
                        array.set(allYs, j, array.get(allYs, j + 1))
                        array.set(allLbs, j, array.get(allLbs, j + 1))
                        array.set(allLns, j, array.get(allLns, j + 1))
                        array.set(allYs, j + 1, ty)
                        array.set(allLbs, j + 1, tlb)
                        array.set(allLns, j + 1, tln)
        if n > 0
            for i = 0 to n - 1
                f_pinOne(array.get(allLbs, i), array.get(allLns, i), array.get(allYs, i))
        // Re-park any unmerged cash-out tags far right (never on RND/OPEN lane)
        if not na(drawUpLb) and not na(drawBSL)
            label.set_x(drawUpLb, bar_index + pinOffset + pinStagger * 4)
            label.set_y(drawUpLb, drawBSL + atrVal * 1.2)
            label.set_style(drawUpLb, label.style_label_left)
            if not na(drawUpLn)
                line.set_extend(drawUpLn, extend.none)
                line.set_x2(drawUpLn, bar_index + pinOffset + pinStagger * 4)
        if not na(drawDnLb) and not na(drawSSL)
            label.set_x(drawDnLb, bar_index + pinOffset + pinStagger * 4)
            label.set_y(drawDnLb, drawSSL - atrVal * 1.2)
            label.set_style(drawDnLb, label.style_label_left)
            if not na(drawDnLn)
                line.set_extend(drawDnLn, extend.none)
                line.set_x2(drawDnLn, bar_index + pinOffset + pinStagger * 4)
    true
